#. Terminal 6:
```
python3 -c "
import rclpy, numpy as np
from rclpy.node import Node
from sensor_msgs.msg import Image
from cv_bridge import CvBridge
rclpy.init(); n=Node('grab'); b=CvBridge(); got=[]
n.create_subscription(Image,'/camera/camera/aligned_depth_to_color/image_raw',                                           
                      lambda m: got.append(b.imgmsg_to_cv2(m,'passthrough')),10)
while not got: rclpy.spin_once(n)
np.save('/tmp/depth.npy', got[0]); print('saved', got[0].shape, got[0].dtype)"
```

```
python3 -c "
import numpy as np
d = np.load('/tmp/depth.npy').astype(np.float32)
d = d/1000.0 if d.max()>100 else d
print('shape', d.shape, 'dtype', np.load('/tmp/depth.npy').dtype)                                                 
print('valid (>0) fraction:', (d>0).mean())
print('depth range (valid):', d[d>0].min(), d[d>0].max())
print('median depth:', np.median(d[d>0]))"
```
